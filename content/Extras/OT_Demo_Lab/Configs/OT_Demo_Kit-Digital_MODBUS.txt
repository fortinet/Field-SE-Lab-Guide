config system global
    set admintimeout 480
    set alias "FortiGateRugged"
    set gui-auto-upgrade-setup-warning disable
    set hostname "FGR"
    set internet-service-database standard
    set switch-controller enable
end

config system interface
    edit "OT_VLAN"
        set vdom "root"
        set ip 192.168.1.1 255.255.255.0
        set allowaccess ping https ssh http
        set description "** MODBUS Devices **"
        set device-identification enable
        set role lan
        set snmp-index 24
        set switch-controller-access-vlan enable
        set interface "fortilink"
        set vlanid 100
    next
end

config ips global 
   set exclude-signatures none 
end

config system settings
    set gui-ips enable
end

config firewall address
    edit "OT_VLAN address"
        set type interface-subnet
        set subnet 192.168.1.0 255.255.255.0
        set interface "OT_VLAN"
    next
    edit "White_Modbus_Device"
        set associated-interface "OT_VLAN"
        set subnet 192.168.1.20 255.255.255.255
    next
    edit "Red_Modbus_Device"
        set associated-interface "OT_VLAN"
        set subnet 192.168.1.10 255.255.255.255
    next
end

config ips sensor
    edit "OT_IPS"
        set block-malicious-url enable
        set scan-botnet-connections block
        config entries
            edit 1
            next
        end
    next
end

config firewall policy
    edit 1
        set name "Intra_OT_VLAN_All_Colors"
        set srcintf "OT_VLAN"
        set dstintf "OT_VLAN"
        set action accept
        set srcaddr "all"
        set dstaddr "all"
        set schedule "always"
        set service "ALL"
        set utm-status enable
        set ssl-ssh-profile "certificate-inspection"
        set av-profile "default"
        set ips-sensor "OT_IPS"
        set application-list "default"
        set logtraffic all
        set auto-asic-offload disable
		set comments "Allow all Writes to Modbus Coils"
    next
    edit 2
        set status disable
        set name "Intra_OT_VLAN_Block_Green"
        set srcintf "OT_VLAN"
        set dstintf "OT_VLAN"
        set action accept
        set srcaddr "all"
        set dstaddr "all"
        set schedule "always"
        set service "ALL"
        set utm-status enable
        set ssl-ssh-profile "certificate-inspection"
        set av-profile "default"
        set ips-sensor "OT_IPS"
        set application-list "Modbus_Block_Blue"
        set logtraffic all
        set auto-asic-offload disable
        set comments "Block Modbus Writes to Green Coil 1"
    next
end

config system proxy-arp
    edit 1
        set interface "OT_VLAN"
        set ip 192.168.1.1
        set end-ip 192.168.1.254
    next
end
